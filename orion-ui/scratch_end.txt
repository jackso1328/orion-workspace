import json
import re

history = []

with open(r'C:\Users\proud\.gemini\antigravity-ide\brain\b17881b6-c3cf-4f45-9147-6e0a0680a957\.system_generated\logs\transcript_full.jsonl', 'r', encoding='utf-8') as f:

                for tool in data.get('tool_calls', []):
                    # write_to_file
                    if tool['name'] == 'write_to_file' and 'App.css' in tool['args'].get('TargetFile', ''):
                        if tool['args'].get('Overwrite'):
                            history.append(('OVERWRITE', tool['args']['CodeContent']))

                    elif tool['name'] == 'replace_file_content' and 'App.css' in tool['args'].get('TargetFile', ''):
                        history.append(('REPLACE', tool['args'].get('ReplacementContent', '')))
                        
                    # run_command
                    elif tool['name'] == 'run_command':
                        cmd = tool['args'].get('CommandLine', '')
                        if 'App.css' in cmd:
                            history.append(('COMMAND', cmd))
        except Exception as e:
            pass

with open('css_history.txt', 'w', encoding='utf-8') as out:
    for i, (action, content) in enumerate(history):
        out.write(f'--- ACTION {i}: {action} ---\n')
        out.write(content + '\n\n')
